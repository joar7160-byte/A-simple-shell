
# P3 batch tests: run ./mysh mysh_tests.txt

# 1. Basic commands, arguments, paths, and comments.
echo TEST_1_basic_commands
echo hello world
/bin/echo absolute_path_works
echo comment_works # This text must not be printed.

# 2. Built-ins. pwd prints the test directory; which prints executable paths.
echo TEST_2_builtins
pwd
which echo cat wc
which /bin/echo
which which

# 3. Redirection. Expected: alpha, then alpha and beta, then replaced.
echo TEST_3_redirection
echo alpha > p3_output.txt
cat < p3_output.txt
echo beta >> p3_output.txt
cat p3_output.txt
echo replaced > p3_output.txt
cat p3_output.txt
# Both forms should create files containing replaced.
cat < p3_output.txt > p3_copy_a.txt
cat > p3_copy_b.txt < p3_output.txt
cat p3_copy_a.txt p3_copy_b.txt
# Verify redirection also works for built-ins.
pwd > p3_pwd.txt
cat p3_pwd.txt
which echo > p3_which.txt
cat p3_which.txt

# 4. Pipelines. Word counts should be 3 and 4 (spacing may vary).
echo TEST_4_pipelines
echo one two three | wc -w
echo one two three four | cat | wc -w
pwd | cat
# Larger output checks concurrent execution and pipe descriptor closure.
seq 1 20000 | cat | wc -l

# 5. Wildcards: multiple patterns, subdirectories, hidden and unmatched files.
echo TEST_5_wildcards
mkdir p3_globs
touch p3_globs/apple.txt p3_globs/apricot.txt p3_globs/banana.log
touch p3_globs/.hidden.txt
echo p3_globs/*.txt
# Previous line: apple.txt and apricot.txt; must exclude .hidden.txt.
echo p3_globs/a*.txt p3_globs/*.log
echo p3_globs/.h*.txt
echo p3_globs/no_match_12345*.xyz
# Last pattern must be printed unchanged.

# 6. cd changes the shell directory; cd in a pipe must not change the parent.
echo TEST_6_cd
pwd
cd p3_globs
pwd
echo *.txt
cd ..
cd p3_globs | cat
pwd
# Previous pwd should still show the original test directory.
